cmake_minimum_required(VERSION 3.22)
project(OpenVSTHostProbe VERSION 0.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(../../research/JUCE juce)
juce_add_console_app(OpenVSTHostProbe PRODUCT_NAME "OpenVST Host Probe")
target_sources(OpenVSTHostProbe PRIVATE main.cpp)
target_compile_definitions(OpenVSTHostProbe PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_LADSPA=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(OpenVSTHostProbe PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
