cmake_minimum_required(VERSION 3.22)
project(OpenVSTWindowsProbe VERSION 0.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
if(NOT DEFINED OPENVST_RUNTIME OR NOT EXISTS "${OPENVST_RUNTIME}")
  message(FATAL_ERROR "OPENVST_RUNTIME must point at an installed CmajPerformer.dll")
endif()
if(NOT DEFINED OPENVST_FIXTURES OR NOT EXISTS "${OPENVST_FIXTURES}")
  message(FATAL_ERROR "OPENVST_FIXTURES must point at the verified fixture directory")
endif()
if(NOT DEFINED OPENVST_JUCE OR NOT EXISTS "${OPENVST_JUCE}/modules/juce_audio_processors")
  message(FATAL_ERROR "OPENVST_JUCE must point at the JUCE 9.0.5 source tree")
endif()
if(NOT DEFINED OPENVST_CMAJOR OR NOT EXISTS "${OPENVST_CMAJOR}/include/cmajor/API/cmaj_Engine.h")
  message(FATAL_ERROR "OPENVST_CMAJOR must point at the Cmajor 1.0.3209 source tree")
endif()
# Windows paths must reach the C++ source with forward slashes: backslashes in a
# string literal would be read by the compiler as escape sequences.
file(TO_CMAKE_PATH "${OPENVST_RUNTIME}" OPENVST_RUNTIME_FWD)
file(TO_CMAKE_PATH "${OPENVST_FIXTURES}" OPENVST_FIXTURES_FWD)
add_subdirectory("${OPENVST_JUCE}" "${CMAKE_BINARY_DIR}/juce-build" EXCLUDE_FROM_ALL)
# Compile the JUCE modules once, as in the verified Linux build, instead of
# duplicating them inside each plugin target.
add_library(openvst_core STATIC src/plugin/Processor.cpp)
target_include_directories(openvst_core PUBLIC src/plugin "${OPENVST_JUCE}/modules" PRIVATE "${OPENVST_CMAJOR}/include")
target_compile_definitions(openvst_core PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 PRIVATE CMAJOR_DLL=1 "OPENVST_RUNTIME=\"${OPENVST_RUNTIME_FWD}\"" "OPENVST_FIXTURES=\"${OPENVST_FIXTURES_FWD}\"")
target_link_libraries(openvst_core PRIVATE juce::juce_audio_processors juce::juce_cryptography)
if(MSVC)
  target_compile_options(openvst_core PUBLIC /utf-8 /W3)
endif()
target_compile_definitions(openvst_core INTERFACE JUCE_MODULE_AVAILABLE_juce_audio_processors=1 JUCE_MODULE_AVAILABLE_juce_gui_extra=1 JUCE_MODULE_AVAILABLE_juce_gui_basics=1 JUCE_MODULE_AVAILABLE_juce_graphics=1 JUCE_MODULE_AVAILABLE_juce_events=1 JUCE_MODULE_AVAILABLE_juce_core=1 JUCE_MODULE_AVAILABLE_juce_data_structures=1 JUCE_MODULE_AVAILABLE_juce_audio_basics=1 JUCE_MODULE_AVAILABLE_juce_audio_processors_headless=1 JUCE_MODULE_AVAILABLE_juce_cryptography=1)
foreach(kind IN ITEMS FX Instrument)
  if(kind STREQUAL "Instrument")
    set(synth TRUE)
    set(code OvIn)
  else()
    set(synth FALSE)
    set(code OvFx)
  endif()
  juce_add_plugin(OpenVST${kind} COMPANY_NAME "OpenVST Private" BUNDLE_ID "org.openvst.private.${kind}" PLUGIN_MANUFACTURER_CODE Ovst PLUGIN_CODE ${code} FORMATS VST3 PRODUCT_NAME "OpenVST ${kind}" IS_SYNTH ${synth} NEEDS_MIDI_INPUT ${synth} NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST FALSE)
  target_sources(OpenVST${kind} PRIVATE src/plugin/Entry.cpp)
  target_compile_definitions(OpenVST${kind} PRIVATE OPENVST_IS_INSTRUMENT=$<BOOL:${synth}>)
  target_link_libraries(OpenVST${kind} PRIVATE openvst_core)
endforeach()
